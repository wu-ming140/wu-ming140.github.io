此外还有关于shell命令行的一些学习
~~~
man  查询说明  man date  
或者加上--help
tab自动补全
path是显示文件夹
which  查命令在清单的哪个文件夹

包括说ls  就在固定文件夹 叫PATH
一般加$PATH   
which ls   #在哪个文件夹找到的"ls"




pwd #显示目前在那个目录/文件夹
ls  #显示文件夹     或者单独在文件夹内ls  显示所有文件
cd  文件夹名字 #进入目录
cd .. #返回上一层
.和..   .是当前目录   ..父目录

cd ~ #回到用户文件夹目录
mkdir 111   #创建文件夹111
touch 222.txt  #创建空文件
rm 222.txt   #删除文件


cat 111.txt    #看文件内容
echo "111" > 111.txt   #写入文件
echo "222" >> 111.txt    #追加一行    单独>是覆盖  
cat hello.txt：打印 hello.txt 的全部内容
sort hello.txt：将 hello.txt 的各行按字典序排序后输出
uniq hello.txt：去除 hello.txt 中连续重复的行
head hello.txt 和 tail hello.txt  打印 hello.txt 的前几行和后几行


#搜索  grep
grep "111" 11.txt #在11.txt中搜索包含111这一行的话  其他不显示
grep -n "1" 111.txt  #-n   等会输出的时候会显示"1"在第几行
grep "error" 日志.txt   #方便查找错误信息

#管道|   前一个命令的输出  给后一个命令当输入
ls | grep "txt"   #筛选文件夹中包含txt的文件
cat 111.txt | grep "1"   #在txt中找到包含1的输出
#这里和grep "1" 111.txt  一样

ls | grep "txt" | wc -l   #数行数   列出文件  筛选出txt的   数个数

保存退出  ctrl+ o   回车
ctrl+x退出

-r  #多数情况是递归(具体要看跟在哪个命令后面)   连同文件夹里的全部东西都处理
rm -r 我的练习        # 删文件夹，连同里面的所有文件
cp -r 文件夹A 文件夹B   # 复制文件夹，连同里面所有内容

rm 笔记.txt     # 删文件，可以
rm 我的练习     # 删文件夹，报错："这是一个目录"
rm -r 我的练习  # 加 -r，才删得掉    不进回收站  彻底删

ls -r      # 这里的 -r 是 reverse（倒序排列），不是递归
grep -r    # 这里的 -r 是递归搜索整个文件夹


sed替换
sed 's/旧/新/' 文件名      #s/  表示替换   /分隔符

cd ~/我的练习
echo "我喜欢苹果" > fruit.txt
echo "苹果很好吃" >> fruit.txt
sed 's/苹果/香蕉/' fruit.txt   #把全部苹果换香蕉
#加上/g表示整一行的全部替换   没有g没行只换第一个

-i直接改文件   不改前要先备份   
sed -i.bak '111' 文件名   #留备份

sed -i 's/苹果/香蕉/g' fruit.txt
cat fruit.txt        # 全变了
